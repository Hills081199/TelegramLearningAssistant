Thêm agent mới — agents/learning_agents.py 
# ✅ Để thêm 1 agent mới, chỉ cần viết class này:

class DockerAgent(BaseLearningAgent):
    agent_id = "docker"

    @property
    def system_prompt(self) -> str:
        return """You are Docker & DevOps Expert 🐳
        Containers, Kubernetes, CI/CD, GitHub Actions..."""

# Thêm vào config/settings.py:
AGENTS_CONFIG["docker"] = AgentConfig(
    name="Docker Expert",
    knowledge_base_path="knowledge_bases/docker",
    use_web_search=True,
    emoji="🐳"
)

# Thêm vào registry:
AGENT_CLASSES["docker"] = DockerAgent

# Done! Agent tự động xuất hiện trong Telegram menu 🎉
